```text

实施前在本项目获准的新分支／独立 worktree 中冻结迁移基线；有既有未提交内容时，由负责人明确哪些需带入或保留，不能只从 HEAD 新建干净树就当原工作已保留。保存原始差异、未跟踪文件与文件模式；只回退本轮迁移差异，已提交时仅反转精确迁移提交并核冲突，不整树恢复。
```

### Windows／PowerShell 7 只读入口示例

先由采用方会话确认项目 Git 根，并保持当前目录在该根；下段不初始化仓库、不迁移或删除文件，不把环境变量写入项目配置。普通终端／CI 没有 DSH 会话环境时停止，由采用方显式提供同一固定安装的 Node、CLI 与锁定证据，不能猜维护者的 profile。

```powershell
if (-not $env:DSH_PROFILE_DIR) { throw '缺 DSH_PROFILE_DIR；先确认采用方所用 profile' }
$profileDir = (Resolve-Path -LiteralPath $env:DSH_PROFILE_DIR -ErrorAction Stop).Path
$packageDir = Join-Path $profileDir 'node_modules/dsh-project-std'
$cliPath = Join-Path $packageDir 'lib/cli.js'
$nodeChoices = @()
if ($env:DSH_HOME) { $nodeChoices += Join-Path $env:DSH_HOME 'dsh-runtimes/dsh-primary-runtime/dependencies/node/bin/node.exe' }
if ($env:USERPROFILE) { $nodeChoices += Join-Path $env:USERPROFILE '.dsh/dsh-runtimes/dsh-primary-runtime/dependencies/node/bin/node.exe' }
$nodePath = $nodeChoices | Where-Object { Test-Path -LiteralPath $_ -PathType Leaf } | Select-Object -First 1
if (-not $nodePath -or -not (Test-Path -LiteralPath $cliPath -PathType Leaf)) { throw '缺可运行 Node 或 CLI，不继续' }
$vText = & $nodePath --version
if ($LASTEXITCODE -ne 0) { throw 'Node 运行失败' }
$v = [version]($vText.TrimStart('v'))
if (-not (($v.Major -eq 22 -and $v.Minor -ge 19) -or $v.Major -ge 24)) { throw 'Node 不满足包 engines' }
$rootPath = (Get-Location).Path
$gitRoot = & git rev-parse --show-toplevel
if ($LASTEXITCODE -ne 0 -or (Resolve-Path -LiteralPath $gitRoot).Path -ne $rootPath) { throw '请在已确认的项目 Git 根运行' }
$commit = 'a035d86bb38b9450b11fc42a1625dfcbc92cf227'
$manifest = Get-Content -LiteralPath (Join-Path $profileDir 'package.json') -Raw | ConvertFrom-Json
if (-not ([string]$manifest.dependencies.'dsh-project-std').EndsWith('#' + $commit)) { throw 'profile 未钉指定完整提交' }
$expectedRaw = @{
  'lib/cli.js' = '59476d5423505be5585d366ed939742db8bea973224dd01d34ffdccc9232470b'
  'lib/migrate.js' = 'e52648b5f4d4e4f788f79e4dc2f869a7e4fab62963f09be38e7d5d491d09851b'
  'lib/main.js' = 'ca60e4e055224af93c8fdb291b648da179286b8c6beaeb4fba7078c60510d3ce'
}
foreach ($rel in $expectedRaw.Keys) {
  if ((Get-FileHash -LiteralPath (Join-Path $packageDir $rel) -Algorithm SHA256).Hash -ne $expectedRaw[$rel]) { throw "制品字节不符：$rel" }
}
$beforeText = & $nodePath $cliPath $rootPath --json
$beforeExit = $LASTEXITCODE
if ($beforeExit -notin @(0, 1, 2)) { throw '检查进程异常退出' }
$findings = @($beforeText -join "`n" | ConvertFrom-Json -ErrorAction Stop)
$identity = @($findings | Where-Object { $_.check -eq 'tool-identity' })
if ($identity.Count -ne 1 -or $identity[0].status -ne 'SKIP') { throw '身份项缺失或无效' }
$actualDigest = [regex]::Match($identity[0].evidence, '(?m)^tool_identity_digest: ([0-9a-f]{64})\r?$').Groups[1].Value
if ($actualDigest -ne '385b78fa54c1684c7e53d5512adc183d83d5324af7a6f361f7e3409dcbb5955a') { throw '完整工具摘要不符' }
if (@($findings.id | Sort-Object -Unique).Count -ne $findings.Count) { throw 'Finding id 重复' }
$migrateText = & $nodePath $cliPath migrate $rootPath
$migrateTextExit = $LASTEXITCODE
$migrateJsonText = & $nodePath $cliPath migrate $rootPath --json
$migrateJsonExit = $LASTEXITCODE
if ($migrateTextExit -ne 0 -or $migrateJsonExit -ne 0) { throw '迁移方案生成失败' }
$migrationPlan = $migrateJsonText -join "`n" | ConvertFrom-Json -ErrorAction Stop
[pscustomobject]@{ Node = $vText; Root = $rootPath; Digest = $actualDigest; CheckExit = $beforeExit; MigrateExit = $migrateJsonExit; Coverage = $migrationPlan.coverage }
```

还须核 `pnpm-lock.yaml` 的 `importers['.'].dependencies['dsh-project-std']`：`specifier` 精确等于 profile manifest 的目标依赖；`version` 所指 locator 能定位到唯一 `packages['dsh-project-std@' + version]` 项（锁格式若不同先查实际格式，不能猜）；该项 `resolution.commit` 精确等于完整提交，`resolution.repo` 与已核来源一致，并有对应 snapshot。只保存这些目标字段及锁文件 SHA-256，不能只在整份锁文件搜索 SHA，也不能输出其它敏感配置。通过技能工具核五个 `std-*` 技能在该采用方会话的实际可用性。上段只读示例没有代做这些检查，不能据末行输出宣布迁移准入。`$beforeText`／`$beforeExit`、`$migrateText`／退出码、`$migrateJsonText`／退出码、`$rootPath`、当前日期及配置摘要须存到采用方确认的**项目外**证据目录，使用 `Set-Content -Encoding utf8`；不要覆盖已有证据。后续 after 用同一组 Node／CLI／配置重新采集；报告 JSON 的 `tool_identity_files` 内逐文件摘要按 LF 归一化，与上面的原始文件 SHA-256 不混用。

### 项目外证据的最小保存示例

承接上面的同一个 PowerShell 会话。先由采用方将 `$evidenceParent` 设为已确认存在、位于项目外的证据父目录；不自动猜用户目录、不把业务样本外传。以下只创建新的专属子目录，不覆盖既有证据；再次执行必须新建一轮证据，不能复用目录。after 仍按项目自己的迁移流程采集和对账，不能复跑本段冒充 after。

```powershell
if (-not $evidenceParent) { throw '先设置采用方确认的项目外 evidenceParent' }
$parentPath = (Resolve-Path -LiteralPath $evidenceParent -ErrorAction Stop).Path
$rootFull = [IO.Path]::GetFullPath($rootPath).TrimEnd([IO.Path]::DirectorySeparatorChar)
$parentFull = [IO.Path]::GetFullPath($parentPath).TrimEnd([IO.Path]::DirectorySeparatorChar)
if ($parentFull -eq $rootFull -or $parentFull.StartsWith($rootFull + [IO.Path]::DirectorySeparatorChar, [StringComparison]::OrdinalIgnoreCase)) { throw '证据父目录必须在项目外' }
$evidenceDir = Join-Path $parentPath ('std-migration-' + [guid]::NewGuid().ToString('N'))
if (Test-Path -LiteralPath $evidenceDir) { throw '目标证据已存在，不覆盖' }
$null = New-Item -ItemType Directory -Path $evidenceDir -ErrorAction Stop
$utf8 = [Text.UTF8Encoding]::new($false)
function Write-NewEvidence([string]$name, [string]$text) {
  $target = Join-Path $evidenceDir $name
  $stream = [IO.File]::Open($target, [IO.FileMode]::CreateNew, [IO.FileAccess]::Write, [IO.FileShare]::None)
  try { $bytes = $utf8.GetBytes($text); $stream.Write($bytes, 0, $bytes.Length) } finally { $stream.Dispose() }
}
Write-NewEvidence 'before.json' ($beforeText -join "`n")
Write-NewEvidence 'migrate.txt' ($migrateText -join "`n")
Write-NewEvidence 'migrate.json' ($migrateJsonText -join "`n")
$cfgPaths = @(@('governance/project.yaml', 'governance/project.yml') | ForEach-Object { Join-Path $rootPath $_ } | Where-Object { Test-Path -LiteralPath $_ -PathType Leaf })
if (@($cfgPaths).Count -ne 1) { throw '默认配置需唯一；显式 --config 的情况先按实际配置调整本记录，不猜' }
$configSha = (Get-FileHash -LiteralPath $cfgPaths[0] -Algorithm SHA256).Hash
$receipt = [pscustomobject]@{ CapturedAt = (Get-Date).ToUniversalTime().ToString('o'); Root = $rootPath; Node = $nodePath; NodeVersion = $vText; CLI = $cliPath; Config = $cfgPaths[0]; ConfigSHA256 = $configSha; ToolDigest = $actualDigest; CheckArgs = @($cliPath, $rootPath, '--json'); CheckExit = $beforeExit; MigrateTextExit = $migrateTextExit; MigrateJsonExit = $migrateJsonExit }
Write-NewEvidence 'receipt.json' ($receipt | ConvertTo-Json -Depth 5)
$evidenceDir
```

`CreateNew` 在文件已存在时失败，不以先检查后普通覆盖冒充防覆盖保障。若父目录含 junction／symlink 或自定义配置，不仅凭词法路径判断物理范围，由采用方先确认真实路径；本示例不替项目决定证据权限与留存。锁定条目、授权、原始配置副本和 caller 验证证据另按已确认范围保存，不能只存摘要替代可恢复原配置。

此文只给当前采用方的核验顺序与报告要求；真实项目的业务选择、调用政策、删除授权及验收由其负责人决定。
